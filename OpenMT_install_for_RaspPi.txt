# Instructions for installing and configuring OpenMT on a Raspberry Pi
======================================================================
# NOTE:
Modem types supported: MMDVM, MMDVM_HS, OPENMT, OPENMT_HS and the OpenMT SX1255
radio (through the sx1255-openmt-bridge program). See README.md.
If running M17 service on an MMDVM modem you will need the firmware modified for M17.
Software tested on Raspberry Pi 2 model B and Pi 3 model B.
The service files and the examples below assume the user "pi" and the OpenMT
directory /home/pi/OpenMT. If yours differ, edit the paths in the .service files.

sudo apt update
sudo apt upgrade

sudo raspi-config
# Goto Interface Options/Serial Port
# Login shell = No , Enable serial hardware = Yes
# Select Finish to exit.

# Edit /boot/firmware/cmdline.txt and add dwc_otg.lpm_enable=0, elevator=deadline and noswap.
# Edit /boot/firmware/config.txt and add dtoverlay=miniuart-bt to [all] section.

sudo apt install git
sudo apt install nginx
sudo apt install php-fpm
sudo apt install mariadb-server mariadb-client libmariadbd-dev
sudo apt install php
sudo apt install php-cli
sudo apt install php-mysql
sudo apt install php-curl
sudo apt install build-essential
sudo apt install cmake
sudo apt install pkg-config

# Only for the parts you use:
sudo apt install libssl-dev       # DMR gateway
sudo apt install libgsm1-dev      # FM service: EchoLink gateway (GSM audio)
sudo apt install libasound2-dev   # SX1255 bridge (ALSA audio)
# NOTE: if cmake says it cannot find "mariadb", also install libmariadb-dev.

git clone https://github.com/kd0oss/OpenMT.git

# Set mariadb server to bind to all network interfaces.
sudo nano /etc/mysql/mariadb.conf.d/50-server.cnf
# Change bind-address to 0.0.0.0 from 127.0.0.1

# Configure databases.
sudo mysql -uroot
create database modem_host;
create user 'openmt'@'%' identified by 'dvpass123';
grant all privileges on modem_host.* to 'openmt'@'%';
flush privileges;
quit

# Upload schemas to databases.
cd OpenMT
# Import data into database tables.
# Password is "dvpass123" when asked.
mysql -u openmt -p modem_host < db/openmt.sql
# Tables for the dashboard Services tab (service control and host list updates).
# These files name the database in every statement, so no "modem_host" is needed.
mysql -u openmt -p < db/services.sql
mysql -u openmt -p < db/host_updates.sql

# Install web based dashboard.
# NOTE: dashboard/www also holds updateHosts.php and updateHostsRun.php. They only run from
# the command line (a web request gets "403"), so having them in the web folder is safe.
# Copy the whole folder again after every update of the project.
sudo rm -R /var/www/html
sudo mkdir /var/www/dashboard
sudo cp dashboard/www/* /var/www/dashboard

# Create database permissions file.
sudo nano /etc/openmt.ini
# Add these lines.
[database]
passwd=dvpass123
host=localhost
# Ctrl+x to save and close editor.
# host= is the machine that runs the database (an address or a name). Every program, the dashboard and
# the service manager read it from this file, so a PC that runs only some of the programs sets host= to
# the machine that has the database. That machine's MariaDB must accept it: set bind-address to its LAN
# address in the MariaDB configuration and grant the openmt user access from the other machine.
# No /etc/openmt.ini (another OS, or a machine without that folder)? Every host, gateway and Modem_Host
# takes the database settings on the command line instead:
#     -s <server>   database server (address or name)       -u <user>   database user (default openmt)
#     -p <password> database password                       -c <file>   use another configuration file
#     p25_host -a 192.168.1.50 -s 192.168.1.50 -u openmt -p dvpass123
# The same settings can come from the environment: OPENMT_DB_HOST, OPENMT_DB_USER, OPENMT_DB_PASSWD and
# OPENMT_INI (path of the configuration file). A command line option wins over the environment, and the
# environment wins over the file. A password on the command line is visible to other users in the
# process list for a moment (the program overwrites it after reading it); the environment variable or a
# configuration file with mode 640 is the safer choice, in a systemd service use Environment= or
# EnvironmentFile=. The dashboard (PHP) still reads /etc/openmt.ini.
# The file holds the database password. The web server user (www-data) and the user that runs the
# programs (pi) must be able to read it, nobody else needs to:
sudo chown pi:www-data /etc/openmt.ini
sudo chmod 640 /etc/openmt.ini

# Update Reflectors and DMR Id's.
# This downloads the lists from the Internet and can take several minutes (the IDs the longest).
# A section can be given (p25, nxdn, ysf, m17, dstar or ids); without one, everything is updated.
# Example: php -q updateHosts.php p25
# Later you can do the same from the dashboard (Settings -> Services, "Update host lists").
cd OpenMT
php -q updateHosts.php

# Update /etc/nginx/sites-available/default to look like this:
sudo nano /etc/nginx/sites-available/default
# NOTE: In the line starting with fastcgi_pass change php version to match installed version.
server {
        listen 80 default_server;
        listen [::]:80 default_server;

        # SSL configuration
        #
        # listen 443 ssl default_server;
        # listen [::]:443 ssl default_server;
        #
        # Note: You should disable gzip for SSL traffic.
        # See: https://bugs.debian.org/773332
        #
        # Read up on ssl_ciphers to ensure a secure configuration.
        # See: https://bugs.debian.org/765782
        #
        # Self signed certs generated by the ssl-cert package
        # Don't use them in a production server!
        #
        # include snippets/snakeoil.conf;

        root /var/www/dashboard;

        # Add index.php to the list if you are using PHP
        index dashboard.html dashboard.htm;

        server_name _;

        location / {
                # First attempt to serve request as file, then
                # as directory, then fall back to displaying a 404.
                try_files $uri $uri/ =404;
        }

        # pass PHP scripts to FastCGI server
        #
        location ~ \.php$ {
                include snippets/fastcgi-php.conf;

                # With php-fpm (or other unix sockets):
                fastcgi_pass unix:/run/php/php8.4-fpm.sock;
                # With php-cgi (or other tcp sockets):
             #   fastcgi_pass 127.0.0.1:9000;
        }

        # deny access to .htaccess files, if Apache's document root
        # concurs with nginx's one
        #
        location ~ /\.ht {
                deny all;
        }
}

# Enable mysql by editing /etc/php/{PHP version}/fpm/php.ini
sudo nano /etc/php/{PHP version}/fpm/php.ini
# Search for extension=mysqli and uncomment it by removing the leading semicolon.
# Check that exec is NOT in the disable_functions line (the dashboard needs exec() to
# start the host list update). It is empty in a default Raspberry Pi OS installation.
# Save changes.

# Now reboot.

# Login to RaspPI and cd to OpenMT.
cd OpenMT

######## NOTE: As you enter each executable directory compile using. #####
mkdir build
cd build
cmake ..
make
##########################################################################
# The programs, and the directory each one is built in:
#   modem/                         Modem_Host                (every modem needs it)
#   modem/sx1255-openmt-bridge/    bridge                    (only for the SX1255 radio)
#   protocols/DMR/                 dmr_host        protocols/DMR/gateway/     dmr_gateway
#   protocols/DSTAR/               dstar_host      protocols/DSTAR/gateway/   dstar_gateway
#   protocols/P25/                 p25_host        protocols/P25/gateway/     p25_gateway
#   protocols/m17/                 m17_host        protocols/m17/gateway/     m17_gateway
#   protocols/NXDN/                nxdn_host       protocols/NXDN/gateway/    nxdn_gateway
#   protocols/YSF/                 ysf_host        protocols/YSF/gateway/     ysf_gateway
#   protocols/FM/                  fm_host         protocols/FM/gateway/      fm_gateway
#   tools/service_manager/         service_manager          (dashboard service control, see below)
# Build only the ones you need.

# Then cd to OpenMT/modem and start modem host using
cd OpenMT/modem/build
sudo ./Modem_Host   # NOTE: sudo needed for access to /dev/ttyAMA0

# If setup properly the host should report no errors.
# You can optionally add -x at startup to display more debug info.
sudo ./Modem_Host -x

# If all is well you will want to start the host in daemon mode to free up the terminal.
ctrl+c to exit host
sudo ./Modem_Host -d
The initial run of the host will populate the database with default configuration.
# Every program takes  -m <modem number 1-10>  (default 1) so that several modems can run
# on one Pi, -d (daemon), -x (debug output) and -v (version).
# Every host and gateway also takes  -a <address>  to say where the program it talks to runs. The
# default is 127.0.0.1 (this machine). A dotted address or a host name will do:
#     p25_host -a 192.168.1.50          connect to a Modem_Host on another machine
#     p25_gateway -a radio.local        connect to the P25 host on another machine
# Hosts connect to Modem_Host, gateways connect to their own host. Modem_Host and the hosts already
# listen on all addresses; open the ports in a firewall only on a network you trust, because the
# links between the programs are not encrypted or authenticated.

# Only for the OpenMT SX1255 radio: the bridge program talks to the radio and gives the modem host
# a serial port (a PTY). See modem/sx1255-openmt-bridge/INSTALL_SERVICE.txt to run it as a service,
# and set the modem host's serial port (modemtty) to /dev/sx1255modem.

# Point a browser to the RaspPI address and confirm Dashboard is accessable.
# Select Settings and a login box should appear. User name = admin, password = admin.
# Update info in General and Modem tabs to match your setup.
# Ignore modem config error, if encountered, for now.
# Restart Modem_Host to use update configuration.
# You will need to do this for each protocol service after running for fist time.

# Now to start M17 mode cd back to OpenMT directory then to protocols/m17 sub-directory.
cd ..
cd protocols/m17/build

# Start M17 host service. Same start up flags apply as modem host.
./m17_host

# If no errors stop and run as daemon.
ctrl+c to exit service
./m17_host -d

# Start gateway if desired. Save start up flags apply.
cd ../gateway/build
./m17_gateway -d

# M17 mode should now appear in the Dashboard.
# If reflectors have be loaded into database you can go to Settings/Mode for link/unlink.

# The other protocols are started the same way: DMR (dmr_host, dmr_gateway), DSTAR (dstar_host,
# dstar_gateway), P25 (p25_host, p25_gateway), NXDN (nxdn_host, nxdn_gateway), YSF (ysf_host,
# ysf_gateway) and FM (fm_host, fm_gateway).
# NOTE: DSTAR gateway requires IRCDDBGateway package from G4KLX GitHub
#       (3rd_party/startup/ircddbgateway.service shows how to run it as a service).
# NOTE: The FM service with the EchoLink gateway needs an OPENMT type modem, an EchoLink account,
#       libgsm and UDP ports 5198 and 5199 forwarded for incoming calls.
#       See protocols/FM/gateway/README.txt.
# Protocols and gateways can be run concurrently.

######## Starting everything when the Pi starts, and the Services tab ##########
# Each program has a systemd service file next to it (modem/modem-host.service,
# protocols/P25/p25-host.service, protocols/P25/gateway/p25-gateway.service and so on).
# Install the ones you use. Example for the modem host, the P25 host and its gateway:
sudo cp modem/modem-host.service /etc/systemd/system/
sudo cp protocols/P25/p25-host.service /etc/systemd/system/
sudo cp protocols/P25/gateway/p25-gateway.service /etc/systemd/system/
# Edit the paths and the User= line in each file if OpenMT is not in /home/pi/OpenMT.
# The user must be allowed to use the serial port (sudo usermod -aG dialout pi) or the modem host
# service can run as root. A gateway needs its host and a host needs the modem host: the files say
# Requires=, so starting a gateway starts everything below it.
sudo systemctl daemon-reload
sudo systemctl enable --now modem-host p25-host p25-gateway

# The dashboard (Settings -> Services) lists these services, starts, stops and restarts them,
# chooses which of them start at boot, and has Restart and Shut down buttons for the Pi.
# The web server is not allowed to control systemd, so a small program running as root does it:
sudo cp tools/service_manager/openmt-service-manager.service /etc/systemd/system/
# (build tools/service_manager first; edit the paths in the file if needed)
sudo systemctl daemon-reload
sudo systemctl enable --now openmt-service-manager
# See tools/service_manager/README.txt. The Services tab warns when this program is not running.
# The list of services is in the database (db/services.sql). To show another one add a row to
# modem_host.services; a service that is not installed is shown as "Not installed".

# Updating the reflector lists and the DMR/NXDN IDs from the dashboard
# (Settings -> Services, "Update host lists"): tick the lists and press "Update selected", or press
# "Update all". The lists offered are the rows of modem_host.host_update_sections (db/host_updates.sql).
# The web server starts dashboard/www/updateHostsRun.php in the background with the "php" program
# (php-cli) and the box shows the progress. Only one update can run at a time.
# If the dashboard reports that it cannot start the update, check that "php -v" works for the
# web server user and that exec() is not disabled in php.ini.
